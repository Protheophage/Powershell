#####################Help and Discovery##################### 
    Update-Help
        #Updates the help files for PowerShell modules on your computer

    Get-Help
        #Displays help about PowerShell cmdlets and concepts
            Get-help Get-*
                #Lists all cmdlets that begin with "Get-"

    Get-Command
        #Gets all commands that are installed on the computer
            Get-Command -Noun *event* 
                    #returns a list of cmdlets dealing with events; the -Noun parameter filters the results to cmdlets with "event" in their name
            Get-Command -Verb Get 
                #returns all cmdlets capable of retrieving things; the -Verb parameter filters the results to cmdlets with "Get" in their name

    Get-Process
        #Gets the processes that are running on the local computer

    Get-Verb
        #Displays a list of approved verbs that can be used in cmdlet names

    Get-PSProvider
        #Displays information about the providers that are available in the current session
    
    Get-ChildItem env:* 
        #Lists all environment variables
    
    Set-Item -Path Env:MyVariable -Value "MyValue" 
        #Sets an environment variable

#####################Properties#####################
    Get-Member
        #Gets the properties and methods of objects
            Get-Process | Get-Member
                #Lists all properties and methods of the process object

    Select-Object
        #Selects specific properties of an object
            Get-Process | Select-Object -Property Name,Id,Path
                #Lists the name, ID, and path of all processes

    Sort-Object
        #Sorts objects by property values
            Get-Process | Sort-Object -Property CPU -Descending
                #Lists all processes sorted by CPU usage in descending order

    Where-Object
        #Filters objects based on specified criteria
            Get-Process | Where-Object {$_.CPU -gt 100}
                #Lists all processes using more than 100 CPU

    Group-Object
        #Groups objects by property values
            Get-Process | Group-Object -Property Name
                #Groups all processes by their name

    Add-Member
        #Adds a property or method to an object
            $process = Get-Process | Select-Object -First 1
            $process | Add-Member -MemberType NoteProperty -Name "NewProperty" -Value "NewValue"
                #Adds a new property called "NewProperty" with the value "NewValue" to the first process object

#####################Formatting Output#####################
    Format-List
        #Formats the output as a list
            Get-Process | Format-List -Property Name,Id,Path
                #Lists the name, ID, and path of all processes in a list format

    Format-Table
        #Formats the output as a table
            Get-Process | Format-Table -Property Name,Id,Path -AutoSize
                #Lists the name, ID, and path of all processes in a table format with auto-sized columns

    Out-GridView
        #Displays the output in a grid view window
            Get-Process | Out-GridView -Title "Processes"
                #Displays all processes in a grid view window with the title "Processes"

    Format-Wide
        #Formats the output in a wide format
            Get-Process | Format-Wide -Property Name -Column 3
                #Lists the names of all processes in a wide format with 3 columns

#####################Format Conversion and File Export#####################
    Convert
        ConvertTo-Csv
            #Converts objects into a series of CSV strings
                Get-Process | ConvertTo-Csv -NoTypeInformation | Out-File -FilePath C:\FT\Processes.csv
            
        ConvertTo-Html
            #Converts objects into a series of HTML strings
                Get-Process | ConvertTo-Html | Out-File -FilePath C:\FT\Processes.html

        ConvertTo-Json
            #Converts objects into a series of JSON strings
                Get-Process | ConvertTo-Json | Out-File -FilePath C:\FT\Processes.json

        ConvertTo-Xml
            #Converts objects into a series of XML strings
                Get-Process | ConvertTo-Xml | Out-File -FilePath C:\FT\Processes.xml

    Export
        Export-Csv
            #Converts objects into a series of CSV strings and saves them to a file
                Get-Process | Export-Csv -Path C:\FT\Processes.csv -NoTypeInformation
            
        Export-Clixml
            #Converts objects into a series of XML strings and saves them to a file
                Get-Process | Export-Clixml -Path C:\FT\Processes.xml

        Out-File
            #Saves the output of a command to a file
                Get-Process | Out-File -FilePath C:\FT\Processes.txt

#####################PS Modules#####################
    Get-Module
        #Get all modules that are currently loaded in the session

    Install-Module
        #Install a module from the PowerShell Gallery
            Install-Module -Name <ModuleName> -Scope CurrentUser -Force -AllowClobber

    New-ModuleManifest
        #Create a module manifest file for a module
            New-ModuleManifest -Path "D:\VsCode\Powershell\FunctionsAndModules\Install_AppFromWeb.psd1" -RootModule "Install_AppFromWeb.psm1" -ModuleVersion "1.0" -Author "Colby Connolly" -Description "Module to install applications from the web"

    Winget
        #Install Winget
            #Bypass Execution Policy for session
            Set-ExecutionPolicy -ExecutionPolicy Bypass -Scope Process -force
            # Install the winget-install script from PowerShell Gallery silently
            Install-Script -Name winget-install -force
            # Run the script with the -Force parameter to install winget silently
            winget-install -Force
        
        #Get list of installed applications/packages
            winget list
        
        #Get available applications/packages that contain a string
            winget show <serch-criteria>
                i.e. winget show office

        #Update all installed applications/packages
            winget upgrade --all

        #Update specific applications/packages
            winget upgrade <app.name>
                i.e. winget upgrade microsoft.office
                    #Note: the app name can be found with winget show

#####################11 Common Parameters#####################
    -Verbose
    Displays detailed information about the operation being performed.

    -Debug
    Provides debugging information during the execution of a command.

    -ErrorAction
    Specifies how the command should respond to errors (e.g., Stop, Continue, SilentlyContinue, etc.).

    -ErrorVariable
    Stores error messages in a specified variable for later use.

    -OutVariable
    Stores output data in a specified variable.

    -OutBuffer
    Specifies the number of objects to buffer before sending them through the pipeline.

    -WarningAction
    Specifies how the command should respond to warnings (e.g., SilentlyContinue, Stop, etc.).

    -WarningVariable
    Stores warning messages in a specified variable for later use.

    -PipelineVariable
    Assigns the current pipeline object to a variable for use within the pipeline.

    -WhatIf
    Simulates the execution of a command, showing what would happen without actually performing the action.

    -Confirm
    Prompts for confirmation before executing a command that could make changes.

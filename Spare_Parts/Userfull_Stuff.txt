#####################Help and Discovery##################### 
Update-Help
Updates the help files for PowerShell modules on your computer

Get-Help
Displays help about PowerShell cmdlets and concepts
    Get-help Get-*
        Lists all cmdlets that begin with "Get-"

Get-Command
Gets all commands that are installed on the computer
    Get-Command -Noun *event* 
            returns a list of cmdlets dealing with events; the -Noun parameter filters the results to cmdlets with "event" in their name
    Get-Command -Verb Get 
        returns all cmdlets capable of retrieving things; the -Verb parameter filters the results to cmdlets with "Get" in their name

Get-Process
Gets the processes that are running on the local computer

Get-Verb
Displays a list of approved verbs that can be used in cmdlet names

    ActiveDirectory
        Get-ADUser
            #Get all properties of all users
                Get-ADUser -Filter *

            #Get the attributes of a user
                get-aduser -Identity <userName> -properties *

            #Get the Password Last Set date
                Get-ADUser -Identity <Username> -properties PwdLastSet | sort Name | ft Name,@{Name='PwdLastSet';Expression={[DateTime]::FromFileTime($_.PwdLastSet)}}

        Set-ADUser
            #Add email alias (SMTP for primary smtp for secondary)
                Set-ADUser -identity <userName> -Add @{Proxyaddresses="SMTP:<userName>@<Domain >.com"}
            
            #Remove email alias
                Set-ADUser -identity <userName> -Remove @{Proxyaddresses="SMTP:<userName>@<Domain >.com"}

        Get-ADDomainController
            #Get information on domain and domain controller
                Get-ADDomainController -Discover -Service PrimaryDC

    GroupPolicy

    Get-Package
        #Get list of installed packages (applications)
            Get-Package -Provider Programs -IncludeWindowsInstaller -ErrorAction SilentlyContinue -Name *

    Uninstall-Package
        #Uninstall a package (application)
            Get-Package -Provider Programs -IncludeWindowsInstaller -ErrorAction SilentlyContinue -Name "sentinel *" | uninstall-package -force

    Winget
        #Install Winget
            #Bypass Execution Policy for session
            Set-ExecutionPolicy -ExecutionPolicy Bypass -Scope Process -force
            # Install the winget-install script from PowerShell Gallery silently
            Install-Script -Name winget-install -force
            # Run the script with the -Force parameter to install winget silently
            winget-install -Force
        
        #Get list of installed applications/packages
            winget list
        
        #Get available applications/packages that contain a string
            winget show <serch-criteria>
                i.e. winget show office

        #Update all installed applications/packages
            winget upgrade --all

        #Update specific applications/packages
            winget upgrade <app.name>
                i.e. winget upgrade microsoft.office
                    #Note: the app name can be found with winget show

#####################11 Common Parameters#####################
    -Verbose
    Displays detailed information about the operation being performed.

    -Debug
    Provides debugging information during the execution of a command.

    -ErrorAction
    Specifies how the command should respond to errors (e.g., Stop, Continue, SilentlyContinue, etc.).

    -ErrorVariable
    Stores error messages in a specified variable for later use.

    -OutVariable
    Stores output data in a specified variable.

    -OutBuffer
    Specifies the number of objects to buffer before sending them through the pipeline.

    -WarningAction
    Specifies how the command should respond to warnings (e.g., SilentlyContinue, Stop, etc.).

    -WarningVariable
    Stores warning messages in a specified variable for later use.

    -PipelineVariable
    Assigns the current pipeline object to a variable for use within the pipeline.

    -WhatIf
    Simulates the execution of a command, showing what would happen without actually performing the action.

    -Confirm
    Prompts for confirmation before executing a command that could make changes.
